        + lineHeight + 14;

      const panelInnerW = infoBoxWidth - panelPad * 2;
      const colMid = textX + Math.floor(panelInnerW * 0.48);
      let textY = infoBoxY + panelPad;

      c.fillStyle = 'rgba(12, 14, 18, 0.97)';
      c.fillRect(infoBoxX, infoBoxY, infoBoxWidth, infoBoxHeight);
      c.strokeStyle = 'rgba(255, 255, 255, 0.18)';
      c.lineWidth = 1;
      c.strokeRect(infoBoxX + 0.5, infoBoxY + 0.5, infoBoxWidth - 1, infoBoxHeight - 1);

      c.textBaseline = 'top';

      c.font = 'bold 11px Arial';

        c.fillStyle = 'rgba(255, 255, 255, 0.07)';
        c.fillRect(textX, textY, panelInnerW, lineHeight);
        c.fillStyle = '#7A8FA0';
        c.font = 'bold 9px Arial';
        c.fillText(title, textX + 4, textY + 2);
        textY += lineHeight + 3;
      };

      const drawPair = (lLabel, lVal, lColor, rLabel, rVal, rColor) => {
        c.font = '9px monospace';
        c.fillStyle = '#778899';
        c.fillText(lLabel, textX, textY);
        if (rLabel) c.fillText(rLabel, colMid, textY);
        c.font = '10px monospace';
        c.fillStyle = lColor || '#E8EEF2';
        c.fillText(lVal, textX, textY + 10);

          c.fillText(rVal, colMid, textY + 10);
        }
        textY += lineHeight + 1;
      };

        const barW = panelInnerW - 52;
        const barX = textX + 34;

        c.fillStyle = '#8899AA';
        c.fillText(String(pct) + '%', textX, textY + 1);
        c.fillStyle = '#2A3038';

        c.font = '9px monospace';

      function smoothCapeValue(prev, target) {
        if (target < prev * 0.1 && prev > 250) {
          return prev + (target - prev) * 0.05;
        }
        return prev + (target - prev) * capeAlpha;
      }
      cs.sb = smoothCapeValue(cs.sb, sbCape);
      cs.mu = smoothCapeValue(cs.mu, muCape);
      cs.ml = smoothCapeValue(cs.ml, mlCape);
      cs.c3 = smoothCapeValue(cs.c3, cape3km);
